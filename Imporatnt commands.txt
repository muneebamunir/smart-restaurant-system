import React, { useEffect, useState } from "react";

function Menu() {
  const [menu, setMenu] = useState([]);

  useEffect(() => {
    fetch("http://localhost:5000/menu")
      .then((res) => res.json())
      .then((data) => {
        console.log("MENU DATA:", data);
        setMenu(data);
      })
      .catch((err) => console.log(err));
  }, []);

  const deleteItem = async (id) => {
    await fetch(`http://localhost:5000/menu/${id}`, {
      method: "DELETE"
    });

    alert("Item deleted");

    setMenu(menu.filter((item) => item._id !== id));
  };

  return (
    <div>
      <h1>Restaurant Menu</h1>

      {menu.map((item) => (
        <div key={item._id}>
          <h3>{item.name}</h3>
          <p>Price: Rs {item.price}</p>
          <p>Category: {item.category}</p>

          <button onClick={() => deleteItem(item._id)}>
            Delete
          </button>
        </div>
      ))}
    </div>
  );
}

export default Menu;

import React, { useEffect, useState } from "react";

function Admin() {
  const [menu, setMenu] = useState([]);
  const [name, setName] = useState("");
  const [price, setPrice] = useState("");
  const [category, setCategory] = useState("");

  // GET MENU ITEMS
  useEffect(() => {
    fetch("http://localhost:5000/menu")
      .then((res) => res.json())
      .then((data) => setMenu(data));
  }, []);

  // ADD ITEM
  const addItem = async () => {
    await fetch("http://localhost:5000/menu", {
      method: "POST",
      headers: {
        "Content-Type": "application/json",
      },
      body: JSON.stringify({ name, price, category }),
    });

    alert("Item Added!");
    window.location.reload();
  };

  // DELETE ITEM
  const deleteItem = async (id) => {
    await fetch(`http://localhost:5000/menu/${id}`, {
      method: "DELETE",
    });

    alert("Item Deleted!");
    window.location.reload();
  };

  return (
    <div>
      <h1>Admin Panel</h1>

      <h2>Add New Food</h2>

      <input
        placeholder="Food Name"
        onChange={(e) => setName(e.target.value)}
      />

      <br /><br />

      <input
        placeholder="Price"
        onChange={(e) => setPrice(e.target.value)}
      />

      <br /><br />

      <input
        placeholder="Category"
        onChange={(e) => setCategory(e.target.value)}
      />

      <br /><br />

      <button onClick={addItem}>Add Item</button>

      <hr />

      <h2>Menu Items</h2>

      {menu.map((item) => (
        <div key={item._id}>
          <h3>{item.name}</h3>
          <p>Price: Rs {item.price}</p>
          <p>Category: {item.category}</p>

          <button onClick={() => deleteItem(item._id)}>
            Delete
          </button>

          <hr />
        </div>
      ))}
    </div>
  );
}

export default Admin;

import { useState } from "react";
import axios from "axios";

function Admin() {

  const [name, setName] = useState("");
  const [price, setPrice] = useState("");
  const [category, setCategory] = useState("");
  const [image, setImage] = useState(null);

  const handleSubmit = async (e) => {
    e.preventDefault();

    const formData = new FormData();
    formData.append("name", name);
    formData.append("price", price);
    formData.append("category", category);
    formData.append("image", image);

    try {

      await axios.post("http://localhost:5000/menu", formData, {
        headers: {
          "Content-Type": "multipart/form-data"
        }
      });

      alert("Menu item added successfully!");

      // clear form
      setName("");
      setPrice("");
      setCategory("");
      setImage(null);

    } catch (error) {
      console.error("Error adding item:", error);
    }
  };

  return (
    <div style={{ padding: "20px" }}>

      <h2>Admin Panel - Add Menu Item</h2>

      <form onSubmit={handleSubmit}>

        <input
          type="text"
          placeholder="Food Name"
          value={name}
          onChange={(e) => setName(e.target.value)}
          required
        />

        <br /><br />

        <input
          type="number"
          placeholder="Price"
          value={price}
          onChange={(e) => setPrice(e.target.value)}
          required
        />

        <br /><br />

        <input
          type="text"
          placeholder="Category"
          value={category}
          onChange={(e) => setCategory(e.target.value)}
          required
        />

        <br /><br />

        <input
          type="file"
          onChange={(e) => setImage(e.target.files[0])}
          required
        />

        <br /><br />

        <button type="submit">Add Food Item</button>

      </form>

    </div>
  );
}

export default Admin;

import React, { useEffect, useState } from "react";
import { Link, useLocation } from "react-router-dom";
import socket from "./socket";

function Menu({ cart, setCart }) {

  const [menuItems, setMenuItems] = useState([]);
  const [search, setSearch] = useState("");
  const [category, setCategory] = useState("All");

  const location = useLocation();
  const params = new URLSearchParams(location.search);

  const tableNumber =
    params.get("table") || localStorage.getItem("tableNumber");

  useEffect(() => {
    if (params.get("table")) {
      localStorage.setItem("tableNumber", params.get("table"));
    }
  }, [location.search]);

  /* ================= IMAGE ================= */

  const getImageUrl = (img) => {
    if (!img) return "https://via.placeholder.com/200";
    return `http://localhost:5000/uploads/${img}`;
  };

  /* ================= FETCH MENU ================= */

  useEffect(() => {
    fetch("http://localhost:5000/menu")
      .then((res) => res.json())
      .then((data) => setMenuItems(data))
      .catch((err) => console.log("Menu fetch error:", err));
  }, []);

  /* ================= WAITER CALL ================= */

  const callWaiter = () => {

    if (!tableNumber) {
      alert("Please scan table QR first");
      return;
    }

    if (!socket || !socket.connected) {
      alert("Connection lost. Please refresh page.");
      return;
    }

    socket.emit("callWaiter", {
      tableNumber,
      time: new Date()
    });

    alert("✅ Waiter is coming to your table!");
  };

  /* ================= CART ================= */

  const addToCart = (item) => {

    const existing = cart.find((i) => i._id === item._id);

    if (existing) {
      const updated = cart.map((i) =>
        i._id === item._id
          ? { ...i, quantity: i.quantity + 1 }
          : i
      );
      setCart(updated);
    } else {
      setCart([...cart, { ...item, quantity: 1 }]);
    }
  };

  /* ================= FILTER ================= */

  const filteredMenu = menuItems.filter((item) => {

    const matchCategory =
      category === "All" || item.category === category;

    const matchSearch =
      item.name.toLowerCase().includes(search.toLowerCase());

    return matchCategory && matchSearch;
  });

  const categoryOrder = [
    "Fast Food",
    "BBQ",
    "Desi",
    "Desi Side Dish",
    "Drinks"
  ];

  const grouped = filteredMenu.reduce((acc, item) => {
    if (!acc[item.category]) acc[item.category] = [];
    acc[item.category].push(item);
    return acc;
  }, {});

  const popularItems = menuItems.slice(0, 3);

  return (
    <div style={{ padding: "30px", background: "#f5f5f5" }}>

      <h1 style={{ textAlign: "center" }}>🍽 Restaurant Menu</h1>

      {tableNumber ? (
        <h3 style={{ textAlign: "center", color: "green" }}>
          🍽 Table: {tableNumber}
        </h3>
      ) : (
        <h3 style={{ textAlign: "center", color: "red" }}>
          Please scan the table QR code
        </h3>
      )}

      {/* CART */}
      <div style={{ textAlign: "center", marginBottom: "10px" }}>
        <Link to={`/cart${tableNumber ? `?table=${tableNumber}` : ""}`}>
          <button>🛒 View Cart ({cart.length})</button>
        </Link>
      </div>

      {/* WAITER BUTTON */}
      <div style={{ textAlign: "center", marginBottom: "20px" }}>
        <button
          onClick={callWaiter}
          style={{
            padding: "10px 20px",
            background: "#007bff",
            color: "white",
            border: "none",
            borderRadius: "8px",
            cursor: "pointer"
          }}
        >
          🔔 Call Waiter
        </button>
      </div>

      {/* SEARCH */}
      <div style={{ textAlign: "center", margin: "20px" }}>
        <input
          type="text"
          placeholder="Search food..."
          value={search}
          onChange={(e) => setSearch(e.target.value)}
          style={{
            padding: "10px",
            width: "300px",
            borderRadius: "8px",
            border: "1px solid #ccc"
          }}
        />
      </div>

      {/* CATEGORY */}
      <div style={{ textAlign: "center", marginBottom: "30px" }}>
        {["All", ...categoryOrder].map((cat) => (
          <button
            key={cat}
            onClick={() => setCategory(cat)}
            style={{
              margin: "5px",
              padding: "10px 15px",
              borderRadius: "20px",
              border: "none",
              background: category === cat ? "#ff6b00" : "#ddd",
              color: category === cat ? "white" : "black",
              cursor: "pointer"
            }}
          >
            {cat}
          </button>
        ))}
      </div>

      {/* POPULAR */}
      <h2>🔥 Popular Items</h2>

      <div style={{
        display: "grid",
        gridTemplateColumns: "repeat(auto-fill,minmax(250px,1fr))",
        gap: "20px"
      }}>
        {popularItems.map((item) => (
          <div key={item._id} style={{
            background: "white",
            borderRadius: "15px",
            overflow: "hidden",
            boxShadow: "0 5px 12px rgba(0,0,0,0.1)"
          }}>

            <img
              src={getImageUrl(item.image)}
              alt={item.name}
              style={{
                width: "100%",
                height: "280px",
                objectFit: "cover",
                display: "block"
              }}
            />

            <div style={{ padding: "15px" }}>
              <h3>{item.name}</h3>
              <h4>Rs {item.price}</h4>

              <button
                onClick={() => addToCart(item)}
                style={{
                  width: "100%",
                  padding: "8px",
                  background: "#ff6b00",
                  color: "white",
                  border: "none",
                  borderRadius: "6px",
                  cursor: "pointer"
                }}
              >
                Add to Cart 🛒
              </button>
            </div>
          </div>
        ))}
      </div>

      {/* CATEGORY ITEMS */}
      {categoryOrder.map((cat) => {
        if (!grouped[cat]) return null;

        return (
          <div key={cat} style={{ marginTop: "40px" }}>
            <h2>{cat}</h2>

            <div style={{
              display: "grid",
              gridTemplateColumns: "repeat(auto-fill,minmax(250px,1fr))",
              gap: "20px"
            }}>
              {grouped[cat].map((item) => (
                <div key={item._id} style={{
                  background: "white",
                  borderRadius: "15px",
                  overflow: "hidden",
                  boxShadow: "0 4px 10px rgba(0,0,0,0.1)"
                }}>

                  <img
                    src={getImageUrl(item.image)}
                    alt={item.name}
                    style={{
                      width: "100%",
                      height: "280px",
                      objectFit: "cover",
                      display: "block"
                    }}
                  />

                  <div style={{ padding: "15px" }}>
                    <h3>{item.name}</h3>
                    <h4>Rs {item.price}</h4>

                    <button
                      onClick={() => addToCart(item)}
                      style={{
                        marginTop: "10px",
                        width: "100%",
                        padding: "10px",
                        background: "#ff6b00",
                        color: "white",
                        border: "none",
                        borderRadius: "8px"
                      }}
                    >
                      Add to Cart 🛒
                    </button>
                  </div>

                </div>
              ))}
            </div>
          </div>
        );
      })}

    </div>
  );
}

export default Menu;
